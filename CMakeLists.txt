cmake_minimum_required(VERSION 3.16)
option(HOST_BUILD "Build offline simulator and tests" OFF)
if(HOST_BUILD)
  project(tesla_unlock_host LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  add_compile_options(-Wall -Wextra -Werror -pedantic)
  include_directories(components/unlock_core/include)
  add_executable(unlock_sim simulator/main.cpp)
  add_executable(core_tests tests/core_tests.cpp)
  enable_testing()
  add_test(NAME safety_state_machine COMMAND core_tests)
else()
  include($ENV{IDF_PATH}/tools/cmake/project.cmake)
  project(tesla_ble_unlock)
endif()
